	if count >= maxPlayers then
		join.Text = "FULL"
		join.BackgroundColor3 = Color3.fromRGB(70,75,85)
		join.AutoButtonColor = false
	else
		join.Text = "JOIN"
		join.BackgroundColor3 = Theme.Main
		join.Activated:Connect(function()
			local TeleportService = game:GetService("TeleportService")
			local ok,err = pcall(function()
				TeleportService:TeleportToPlaceInstance(game.PlaceId,server.id,player)
			end)
			if not ok then
				serverStatus.Text = "Join failed: "..tostring(err)
			end
		end)
	end
end

local function checkPublicServers()
	clearServers()
	serverStatus.Text = "Checking available servers..."
	checkServers.Text = "CHECKING..."
	checkServers.Active = false

	local ok,data = pcall(function()
		local raw = game:HttpGet(
			"https://games.roblox.com/v1/games/"..tostring(game.PlaceId).."/servers/Public?sortOrder=Asc&limit=100"
		)
		return game:GetService("HttpService"):JSONDecode(raw)
	end)

	checkServers.Active = true
	checkServers.Text = "CHECK SERVERS"

	if not ok or type(data) ~= "table" then
		serverStatus.Text = "Could not load servers. Try CHECK SERVERS again."
		return
	end

	local servers = data.data or {}
	local shown = 0
	for _,server in ipairs(servers) do
		if server.id then
			addServerCard(server)
			shown += 1
		end
	end

	serverStatus.Text = "Detected "..tostring(shown).." public server(s). Full servers are marked FULL."
	task.defer(function()
		serverList.Size = UDim2.new(1,0,0,serverLayout.AbsoluteContentSize.Y)
	end)
end

checkServers.Activated:Connect(checkPublicServers)

local transparencyDragging = false

local function setTransparency(x)
	local percent = math.clamp(
		(x-transparencyBar.AbsolutePosition.X) /
		transparencyBar.AbsoluteSize.X,
		0,
		1
	)

	transparencyDot.Position =
		UDim2.new(percent,-7,.5,-7)

	local value = math.floor(percent*100)

	transparencyTitle.Text =
		"TRANSPARENCY "..value.."%"

	main.BackgroundTransparency = percent
end

transparencyDot.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1
	or input.UserInputType == Enum.UserInputType.Touch then
		transparencyDragging = true
	end
end)

UIS.InputChanged:Connect(function(input)
	if transparencyDragging and
		(input.UserInputType == Enum.UserInputType.MouseMovement
		or input.UserInputType == Enum.UserInputType.Touch) then

		setTransparency(input.Position.X)
	end
end)

UIS.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1
	or input.UserInputType == Enum.UserInputType.Touch then
		transparencyDragging = false
	end
end)

local dragging = false
local dragStart
local startPos

title.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.Touch
    or input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = true
        dragStart = input.Position
        startPos = main.Position
    end
end)

UIS.InputChanged:Connect(function(input)
    if dragging and (
        input.UserInputType == Enum.UserInputType.Touch
        or input.UserInputType == Enum.UserInputType.MouseMovement
    ) then
        local delta = input.Position - dragStart
        main.Position = UDim2.new(
            startPos.X.Scale,
            startPos.X.Offset + delta.X,
            startPos.Y.Scale,
            startPos.Y.Offset + delta.Y
        )
    end
end)

UIS.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.Touch
    or input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = false
    end
end)

local miniDragging = false
local miniMoved = false
local miniDragStart
local miniStartPos

mini.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.Touch
	or input.UserInputType == Enum.UserInputType.MouseButton1 then

		miniDragging = true
		miniMoved = false
		miniDragStart = input.Position
		miniStartPos = mini.Position
	end
end)

UIS.InputChanged:Connect(function(input)
	if miniDragging and (
		input.UserInputType == Enum.UserInputType.Touch
		or input.UserInputType == Enum.UserInputType.MouseMovement
	) then

		local delta = input.Position - miniDragStart

		if math.abs(delta.X) > 5 or math.abs(delta.Y) > 5 then
			miniMoved = true
		end

		mini.Position = UDim2.new(
			miniStartPos.X.Scale,
			miniStartPos.X.Offset + delta.X,
			miniStartPos.Y.Scale,
			miniStartPos.Y.Offset + delta.Y
		)
	end
end)

UIS.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.Touch
	or input.UserInputType == Enum.UserInputType.MouseButton1 then

		if miniDragging and not miniMoved then
			main.Visible = true
			mini.Visible = false
		end

		miniDragging = false
		miniMoved = false
	end
end)

	gui:Destroy()
end)
